  47 class Workspace:
  48     def __init__(self, root: Path, task: Task):

  53     def safe_file(self, name: str) -> Path:
  54         if name not in {'rules.json','README.md'}: raise ToolError('path_not_allowed')
  55         p=(self.root/name)
  56         if p.is_symlink(): raise ToolError('symlink_not_allowed')
  57         q=p.resolve()
  58         if q.parent!=self.root: raise ToolError('path_escape')
  59         return q
  60     def rules(self) -> list[str]:
  61         obj=json.loads(self.safe_file('rules.json').read_text(encoding='utf-8'))

  65     def included(self,path: str) -> bool:

  68     def grade(self) -> Json:
  69         d=self.task.excluded
  70         cases=[(d+'/a.txt',False),(d+'/nested/b.txt',False),(d+'2/a.txt',True),('src/a.py',True)]

  78         for k,v in args.items():
  79             if not isinstance(v,str): raise ToolError('invalid_type:'+k)

  82             # Deliberately irrelevant data lets the context exercise remove bytes without losing state.
  83             return {'content':raw.decode('utf-8'),'sha256':digest(raw),'diagnostics':'fixture-noise\n'*120}
  84         if name=='set_exclusion':

 152 def prepare_history(messages:list[Json],policy:str) -> list[Json]:
 153     copied=json.loads(dump(messages))
 154     if policy=='full':return copied

 165 def run_episode(task:Task,backend:Backend|None=None,policy:str='full',max_rounds:int=8) -> Json:
 166     backend=backend or ScriptedBackend()
 167     if max_rounds<1:raise ValueError('max_rounds must be positive')
 168     begin=time.perf_counter_ns();events=[];seen:set[str]=set();req_bytes=0;usages=[];rounds=0;stop='budget_exhausted'
 169     def emit(kind:str,**payload):events.append({'seq':len(events),'at_ms':(time.perf_counter_ns()-begin)/1e6,'event':kind,**payload})
 170     with tempfile.TemporaryDirectory(prefix='ai-course-') as tmp:
 171         w=Workspace(Path(tmp),task)
 172         initial=w.grade();emit('task_started',task_id=task.task_id,split=task.split,backend=backend.kind,initial=initial)
 173         messages=[{'role':'system','content':SYSTEM},{'role':'user','content':task.prompt()}]
 174         for _ in range(max_rounds):
 175             rounds+=1; view=prepare_history(messages,policy);size=len(dump({'messages':view,'tools':TOOLS}).encode());req_bytes+=size
 176             t0=time.perf_counter_ns();emit('model_requested',round=rounds,request_bytes=size)
 177             try:
 178                 response=backend.complete(view,TOOLS);msg=response['message']
 179                 if not isinstance(msg,dict) or msg.get('role')!='assistant':raise ProtocolError('assistant_message_required')
 180                 calls=msg.get('tool_calls') or []
 181                 if not isinstance(calls,list):raise ProtocolError('tool_calls_must_be_array')
 182                 validated=[]
 183                 for call in calls:
 184                     cid=call.get('id');fn=call.get('function',{})
 185                     if not isinstance(cid,str) or not cid or cid in seen or cid in [c[0] for c in validated]:raise ProtocolError('missing_or_duplicate_call_id')
 186                     if call.get('type')!='function' or not isinstance(fn.get('name'),str):raise ProtocolError('unsupported_call_type')
 187                     validated.append((cid,fn['name'],fn.get('arguments','')))
 188             except (BackendError,ProtocolError,KeyError,TypeError,AttributeError) as e:
 189                 emit('model_failed',error=str(e));stop='backend_or_protocol_error';break
 190             emit('model_returned',round=rounds,duration_ms=(time.perf_counter_ns()-t0)/1e6,tool_calls=len(calls),usage=response.get('usage'))
 191             usages.append(response.get('usage'));messages.append(msg)
 192             if not calls:stop='model_final';break
 193             for cid,name,raw in validated:
 194                 seen.add(cid);emit('tool_proposed',call_id=cid,name=name,arguments=raw)
 195                 t1=time.perf_counter_ns()
 196                 try:
 197                     args=json.loads(raw) if isinstance(raw,str) else None
 198                     value=w.execute(name,args);result={'ok':True,'result':value}
